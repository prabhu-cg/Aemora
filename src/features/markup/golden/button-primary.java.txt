package com.example.core.models;

import org.apache.sling.api.resource.Resource;
import org.apache.sling.models.annotations.Default;
import org.apache.sling.models.annotations.DefaultInjectionStrategy;
import org.apache.sling.models.annotations.Model;
import org.apache.sling.models.annotations.injectorspecific.ValueMapValue;

@Model(adaptables = Resource.class, defaultInjectionStrategy = DefaultInjectionStrategy.OPTIONAL)
public class ButtonPrimaryModel {

    @ValueMapValue
    private String icon;

    @ValueMapValue
    @Default(values = "sm")
    private String size;

    @ValueMapValue
    @Default(values = "none")
    private String iconStyle;

    @ValueMapValue
    @Default(values = "Label")
    private String label;

    public String getIcon() {
        return icon;
    }

    public String getSize() {
        return size;
    }

    public String getSizeModifier() {
        return toModifier(size);
    }

    public String getIconStyle() {
        return iconStyle;
    }

    public String getIconStyleModifier() {
        return toModifier(iconStyle);
    }

    public String getLabel() {
        return label;
    }

    private static String toModifier(String value) {
        return value == null ? "" : value.toLowerCase().replaceAll("[^a-z0-9]+", "-").replaceAll("^-+|-+$", "");
    }
}
